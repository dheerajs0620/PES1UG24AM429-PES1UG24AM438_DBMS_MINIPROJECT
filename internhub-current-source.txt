function companyStyle(company: string) {
  const name = company.toLowerCase();

  if (name.includes('google')) return 'google';
  if (name.includes('microsoft')) return 'microsoft';
  if (name.includes('amazon')) return 'amazon';
  if (name.includes('adobe')) return 'adobe';

  return 'default';
}

function companyInitial(company: string) {
  const name = company.toLowerCase();

  if (name.includes('google')) return 'G';
  if (name.includes('microsoft')) return '▦';
  if (name.includes('amazon')) return 'a';
  if (name.includes('adobe')) return 'A';

  return company.trim().charAt(0).toUpperCase() || '?';
}

function formatStipend(value: number | string | null | undefined) {
  const amount = Number(value ?? 0);

  return Number.isFinite(amount) && amount > 0
    ? `₹${amount.toLocaleString('en-IN')}/month`
    : 'Not specified';
}

function durationFor(
  job: Opportunity,
  index: number,
  sample: boolean
) {
  if (job.duration) return job.duration;

  if (job.duration_months) {
    return `${job.duration_months} months`;
  }

  return sample
    ? ['3–6 months', '6 months', '4–6 months', '3 months'][index]
    : 'Not specified';
}

"use client";

import { useState, FormEvent } from "react";
import Link from "next/link";
import { useRouter } from "next/navigation";

export default function SignupPage() {
  const router = useRouter();

  const [form, setForm] = useState({
    firstName: "",
    lastName: "",
    email: "",
    password: "",
    confirmPassword: "",
  });

  const [showPassword, setShowPassword] = useState(false);
  const [showConfirmPassword, setShowConfirmPassword] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  function handleChange(e: React.ChangeEvent<HTMLInputElement>) {
    setForm({ ...form, [e.target.name]: e.target.value });
  }

  async function handleSubmit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setError("");

    if (form.password.length < 8) {
      setError("Password must contain at least 8 characters.");
      return;
    }

    if (form.password !== form.confirmPassword) {
      setError("Passwords do not match.");
      return;
    }

    setLoading(true);

    try {
      const response = await fetch("/api/auth/signup", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        credentials: "include",
        body: JSON.stringify({
          firstName: form.firstName,
          lastName: form.lastName,
          email: form.email,
          password: form.password,
        }),
      });

      const data = await response.json();

      if (!response.ok || !data.success) {
        setError(data.message || "Unable to create your account.");
        return;
      }

      router.push("/dashboard");
      router.refresh();
    } catch {
      setError("Unable to connect to the server. Please try again.");
    } finally {
      setLoading(false);
    }
  }

  const inputStyle =
    "w-full rounded-xl border border-white/10 bg-[#080b14] px-4 py-3 text-sm text-white outline-none transition placeholder:text-gray-600 focus:border-purple-500 focus:ring-2 focus:ring-purple-500/20";

  return (
    <main className="relative flex min-h-screen items-center justify-center overflow-hidden bg-[#080b14] px-5 py-10 text-white">

const statusColors: Record<string, string> = {
  shortlisted: "#16d98b",
  "under review": "#159cf5",
  interview: "#a855f7",
  rejected: "#737b9d",
  applied: "#e6a23c",
  selected: "#16d98b",
  offered: "#16d98b",
};

function normalizeStatus(status?: string) {
  const value = (status || "Applied").toLowerCase().trim();

  if (value.includes("shortlist") || value === "selected" || value === "offered") return "Shortlisted";
  if (value.includes("interview")) return "Interview";
  if (value.includes("review") || value.includes("screen")) return "Under Review";
  if (value.includes("reject")) return "Rejected";

  return "Applied";
}

function CompanyMark({ name }: { name: string }) {
  const initials = name
    .split(/\s+/)
    .filter(Boolean)
    .slice(0, 2)
    .map((part) => part[0])
    .join("")
    .toUpperCase();

  return (
    <div className="flex h-14 w-14 shrink-0 items-center justify-center rounded-2xl border border-fuchsia-400/30 bg-gradient-to-br from-purple-500/25 via-fuchsia-500/10 to-blue-500/20 text-xl font-bold text-fuchsia-200 shadow-[0_0_24px_rgba(168,85,247,0.12)]">
      {initials || "IN"}
    </div>
  );
}

function SectionTitle({
  icon,
  title,
  subtitle,
}: {
  icon: string;
  title: string;
  subtitle?: string;
}) {
  return (

  );
}

          ...authData.user,

            cache: "no-store",
          }),

                          <span key={skill}>{skill}</span>
                        )
                      )}
                    </div>

                    <h2 className="ih-preferred-heading">
                      Preferred Skills
                    </h2>

                      ))}
                    </div>
                  </div>

                      for this opportunity.
                    </p>

                      <strong>
                        {Math.round(getMatchPercent(selectedOpportunity))}%
                      </strong>
                      <span>Overall Match</span>
                    </div>
                  </div>

                </div>
              )}

              {/* ABOUT COMPANY TAB */}
              {detailTab === 'company' && (
                <div className="ih-job-detail-card ih-company-about">
                  <h2>
                    About{' '}
                    {selectedOpportunity.company_name ||
                      selectedOpportunity.company ||
                      'Company'}
                  </h2>

                  <p className="ih-job-description">
                    {selectedOpportunity.company_description ||
                      'Company information has not been provided for this opportunity yet.'}
                  </p>

                  <div className="ih-company-detail-grid">
                    <div>
                      <span>Company</span>
                      <strong>
                        {selectedOpportunity.company_name ||
                          selectedOpportunity.company ||
                          'Not specified'}
                      </strong>
                    </div>

                    <div>
                      <span>Location</span>
                      <strong>
                        {selectedOpportunity.location || 'Not specified'}
                      </strong>
                    </div>

                    <div>
                      <span>Website</span>
                      <strong>
                        {selectedOpportunity.company_website || 'Not provided'}
                      </strong>
                    </div>
                  </div>
                </div>
              )}

            </div>
          </div>
        )}

        {/* APPLICATION FEEDBACK */}
        {applyMsg && (
          <div className="ih-application-feedback">
            {applyMsg}
            <button
              type="button"
              onClick={() => setApplyMsg('')}
            >
              ×
            </button>
          </div>
        )}
                

      </div>
    </main>
  );
}

function formatDate(value?: string) {
  if (!value) return "Date unavailable";

  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return "Date unavailable";

  return date.toLocaleDateString("en-IN", {
    day: "numeric",
    month: "short",
    year: "numeric",
  });
}

function getStatusClass(status: string) {
  switch (status?.toLowerCase()) {
    case "offer":
      return "border-emerald-400/30 bg-emerald-400/10 text-emerald-300";
    case "interview":
      return "border-cyan-400/30 bg-cyan-400/10 text-cyan-300";
    case "assessment":
      return "border-amber-400/30 bg-amber-400/10 text-amber-300";
    case "rejected":
      return "border-rose-400/30 bg-rose-400/10 text-rose-300";
    default:
      return "border-purple-400/30 bg-purple-400/10 text-purple-300";
  }
}
