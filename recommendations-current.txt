'use client';

import React, { useState, useEffect } from 'react';

export default function RecommendationsPage() {
  const [file, setFile] = useState<File | null>(null);
  const [loading, setLoading] = useState(false);
  const [extractedSkills, setExtractedSkills] = useState<string[]>([]);
  const [recommendations, setRecommendations] = useState<any[]>([]);
  const [myApplications, setMyApplications] = useState<any[]>([]);
  const [stipendStats, setStipendStats] = useState<any[]>([]);
  
  // Dynamic Candidate Inputs
  const [candidateName, setCandidateName] = useState('Dheeraj S');
  const [candidateEmail, setCandidateEmail] = useState('dheeraj@pesu.edu');
  
  const [error, setError] = useState('');
  const [applyMsg, setApplyMsg] = useState('');

  // Fetch applications specifically for the currently entered email
  const fetchApplicationsForCandidate = async (emailToFetch: string) => {
    if (!emailToFetch.trim()) {
      setMyApplications([]);
      return;
    }
    try {
      const res = await fetch(`/api/applications?email=${encodeURIComponent(emailToFetch.trim())}`);
      const data = await res.json();
      if (data.success) {
        setMyApplications(data.applications || []);
        setStipendStats(data.stipend_analytics || []);
      }
    } catch (err) {
      console.error('Error fetching candidate applications:', err);
    }
  };

  // Re-fetch whenever candidateEmail changes
  useEffect(() => {
    fetchApplicationsForCandidate(candidateEmail);
  }, [candidateEmail]);

  const handleScanAndRecommend = async () => {
    if (!file) {
      setError('Please select a CV file first.');
      return;
    }

    setLoading(true);
    setError('');

    try {
      let cvText = '';
      if (file.type === 'application/pdf' || file.name.endsWith('.pdf')) {
        // eslint-disable-next-line @typescript-eslint/ban-ts-comment
        // @ts-ignore
        const pdfjsLib = await import('pdfjs-dist/build/pdf');
        pdfjsLib.GlobalWorkerOptions.workerSrc = '/pdf.worker.min.js';

        const arrayBuffer = await file.arrayBuffer();
        const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;

        for (let i = 1; i <= pdf.numPages; i++) {
          const page = await pdf.getPage(i);
          const content = await page.getTextContent();
          const strings = content.items
            .map((item: any) => (typeof item.str === 'string' ? item.str : ''))
            .filter((str: string) => str.length > 0);
          cvText += strings.join(' ') + ' ';
        }
      } else {
        cvText = await file.text();
      }

      const response = await fetch('/api/extract-cv', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ cvText, fileName: file.name }),
      });

      const data = await response.json();
      if (!data.success) throw new Error(data.error || 'Failed to scan CV');

      // Persist the uploaded resume and recognized skills for the logged-in student
      const saveResponse = await fetch('/api/resumes', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          fileName: file.name,
          extractedText: cvText,
          extractedSkills: data.extracted_skills || [],
        }),
      });

      const saveData = await saveResponse.json();

      if (!saveResponse.ok || !saveData.success) {
        throw new Error(saveData.error || 'Failed to save your resume.');
      }

      console.log('Resume saved:', saveData);

      setExtractedSkills(data.extracted_skills || []);
      setRecommendations(data.recommendations || []);
      
      // Refresh applications for this exact candidate
      fetchApplicationsForCandidate(candidateEmail);
    } catch (err: any) {
      setError(err.message || 'An error occurred during CV scanning.');
    } finally {
      setLoading(false);
    }
  };

  const handleApply = async (opp: any) => {
    setApplyMsg('');
    try {
      const res = await fetch('/api/applications', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          opportunity_id: opp.opportunity_id,
          candidate_name: candidateName,
          candidate_email: candidateEmail,
          extracted_skills: extractedSkills,
          match_score: opp.match_score,
        }),
      });

      const data = await res.json();
      if (data.success) {
        setApplyMsg(`Applied to ${opp.title} successfully!`);
        // Dynamically reload application history for this candidate
        fetchApplicationsForCandidate(candidateEmail);
      } else {
        setApplyMsg(`Application failed: ${data.error}`);
      }
    } catch (err: any) {
      setApplyMsg(`Error applying: ${err.message}`);
    }
  };

  return (
    <main className="ih-page">
      <header className="ih-header">
        <a href="/" className="ih-brand">
          <span className="ih-brand-icon">✥</span>
          InternHub
        </a>

        <nav className="ih-nav">
          <a href="/">Home</a>
          <a className="ih-active" href="/recommendations">Jobs</a>
          <a href="/companies">Companies</a>
          <a href="/stipend-insights">Stipend Insights</a>
          <a href="/about">About</a>
        </nav>

        <div className="ih-profile">
          <span className="ih-avatar">
            {(candidateName || "S").charAt(0).toUpperCase()}
          </span>
          <span>Student</span>
        </div>

      <div className="ih-container">
        <section className="ih-hero">
          <div>
            <div className="ih-eyebrow">
              <span className="ih-dot" />
              YOUR CAREER STARTS HERE
            </div>
            <h1>Upload Your Resume</h1>

              Get personalized internship recommendations based on your skills.

            <div className="ih-art-glow" />

              <strong>RESUME</strong>
              <div className="ih-art-avatar">♙</div>
              <i />
              <i />
              <i />
              <i />
            </div>
            <div className="ih-cloud">↑</div>
          </div>

        <section className="ih-steps">

            <span>1</span>
            <small>Upload Resume</small>
          </div>
          <div className="ih-step-line" />

          </div>

            <span>3</span>
            <small>View Recommendations</small>
          </div>
        </section>

        <section className="ih-main-grid">
          <div className="ih-upload-panel">
            <div
              className="ih-dropzone"
              onDragOver={(event) => event.preventDefault()}
              onDrop={(event) => {
                event.preventDefault();
                const droppedFile = event.dataTransfer.files?.[0];
                if (
                  droppedFile &&
                  (droppedFile.name.toLowerCase().endsWith(".pdf") ||
                   droppedFile.name.toLowerCase().endsWith(".txt"))
                ) {
                  setFile(droppedFile);
                  setError("");
                } else if (droppedFile) {
                  setError("Please select a PDF or TXT file.");
                }
              }}
            >
              <div className="ih-upload-symbol">
                <svg viewBox="0 0 48 48" fill="none">
                  <path

                    strokeLinecap="round"
                    strokeLinejoin="round"

                Choose File

                  type="file"
                  accept=".pdf,.txt,application/pdf,text/plain"
                  onChange={(event) => {
                    const selected = event.target.files?.[0] || null;
                    if (selected && selected.size > 5 * 1024 * 1024) {
                      setError("File size must be 5 MB or less.");
                      setFile(null);
                      return;
                    }
                    setFile(selected);
                    setError("");
                  }}
                />
              </label>
            </div>

            {file && (
              <div className="ih-file-card">
                <div className="ih-pdf-icon">
                  {file.name.toLowerCase().endsWith(".pdf") ? "PDF" : "TXT"}
                </div>
                <div className="ih-file-details">
                  <strong>{file.name}</strong>
                  <small>{(file.size / 1024).toFixed(1)} KB</small>
                </div>
                <span className="ih-file-ready">
                  {loading ? "Processing..." : "Ready"}
                </span>
                <button
                  className="ih-remove"
                  onClick={() => setFile(null)}
                  disabled={loading}
                  title="Remove file"
                >
                  ×
                </button>
              </div>
            )}

              <div className="ih-alert ih-success">{applyMsg}</div>
            )}

            <button

                <>

                  Scanning Resume...

                <>✦ Extract Skills from Resume <span>→</span></>
              )}
            </button>

            <p className="ih-privacy">
              <span>▣</span> Your resume is securely processed for personalized matching.
            </p>
          </div>

          <aside className="ih-sidebar">
            <div className="ih-info-card">
              <h3>How it Works?</h3>

              <div className="ih-info-step">
                <span className="ih-info-icon">▤</span>
                <div>
                  <strong>Upload Resume</strong>
                  <p>Upload your latest resume in PDF or TXT format.</p>
                </div>
              </div>

              <div className="ih-connector">↓</div>

              <div className="ih-info-step">
                <span className="ih-info-icon">⚙</span>

                  <strong>Extract Skills</strong>
                  <p>Our scanner identifies your technical skills and expertise.</p>
                </div>
              </div>

              <div className="ih-connector">↓</div>

              <div className="ih-info-step">
                <span className="ih-info-icon">◎</span>
                <div>
                  <strong>Get Recommendations</strong>
                  <p>Discover internships that align with your skill set.</p>
                </div>
              </div>
            </div>

            <div className="ih-info-card ih-features">
              <h3>Supported Features</h3>
              <p><span>✓</span> Automatic skill extraction</p>
              <p><span>✓</span> Domain and role classification</p>
              <p><span>✓</span> Personalized internship matching</p>
              <p><span>✓</span> Match score and skill insights</p>
            </div>
          </aside>
        </section>

        <section className="ih-skills-card">
          <div className="ih-section-heading">
            <div>
              <h2>Extracted Skills Preview</h2>
              <p>
                {extractedSkills.length
                  ? "Skills identified from your uploaded resume"
                  : "Your skills will appear here after resume analysis"}
              </p>
            </div>
            <div className="ih-skill-total">
              <strong>{extractedSkills.length}</strong>
              <small>Skills Found</small>
            </div>
          </div>

          {extractedSkills.length > 0 ? (
            <div className="ih-skills-body">
              <div className="ih-skill-circle">
                <div>
                  <strong>{extractedSkills.length}</strong>
                  <small>Skills Found</small>
                </div>
              </div>

              <div className="ih-skill-list">
                {[
                  {
                    name: "Programming & Development",
                    icon: "</>",
                    matches: ["Python", "Java", "C", "C++", "JavaScript", "TypeScript", "React", "Next.js", "Node.js", "Express.js", "FastAPI"]
                  },
                  {
                    name: "Databases",
                    icon: "▤",
                    matches: ["SQL", "MySQL", "PostgreSQL", "MongoDB", "Neo4j", "Redis", "ChromaDB"]
                  },
                  {
                    name: "AI & Machine Learning",
                    icon: "✺",
                    matches: ["Machine Learning", "Deep Learning", "Computer Vision", "Natural Language Processing", "PyTorch", "TensorFlow", "OpenCV"]
                  },
                  {
                    name: "Tools & Technologies",
                    icon: "⚙",
                    matches: ["Git", "Docker", "Kubernetes", "Linux", "Wazuh"]
                  }
                ].map((group) => {
                  const matched = extractedSkills.filter((skill: string) =>
                    group.matches.some((item) =>
                      item.toLowerCase() === skill.toLowerCase()
                    )
                  );

                  return matched.length ? (
                    <div className="ih-skill-group" key={group.name}>
                      <div className="ih-group-name">
                        <span>{group.icon}</span>
                        {group.name}
                      </div>
                      <div className="ih-tags">
                        {matched.map((skill: string) => (
                          <span key={skill}>{skill}</span>
                        ))}
                      </div>
                    </div>
                  ) : null;
                })}

                {extractedSkills
                  .filter((skill: string) =>
                    ![
                      "Python", "Java", "C", "C++", "JavaScript", "TypeScript",
                      "React", "Next.js", "Node.js", "Express.js", "FastAPI",
                      "SQL", "MySQL", "PostgreSQL", "MongoDB", "Neo4j", "Redis",
                      "ChromaDB", "Machine Learning", "Deep Learning",
                      "Computer Vision", "Natural Language Processing",
                      "PyTorch", "TensorFlow", "OpenCV", "Git", "Docker",
                      "Kubernetes", "Linux", "Wazuh"
                    ].some((item) => item.toLowerCase() === skill.toLowerCase())
                  )
                  .map((skill: string) => (
                    <span className="ih-extra-skill" key={skill}>{skill}</span>
                  ))}
              </div>
            </div>
          ) : (
            <div className="ih-empty-skills">
              <span>✧</span>
              <p>Upload and scan your resume to see your extracted skills.</p>
            </div>
          )}
        </section>

        {recommendations.length > 0 && (
          <section className="ih-results-section">
            <div className="ih-results-heading">
              <div>
                <div className="ih-eyebrow">MATCHED FOR YOU</div>
                <h2>Recommended Opportunities</h2>
                <p>Internships selected based on your extracted skills.</p>
              </div>
              <span className="ih-result-count">{recommendations.length} Matches</span>
            </div>

            <div className="ih-recommendations">
              {recommendations.map((rec) => {
                const isAlreadyApplied = myApplications.some(
                  (app) => app.opportunity_id === rec.opportunity_id
                );

                return (
                  <div key={rec.opportunity_id} className="ih-job-card">
                    <div className="ih-job-details">
                      <h3>{rec.title}</h3>
                      <p className="ih-company">
                        {rec.company_name} · {rec.opportunity_type}
                      </p>
                      <p>{rec.description}</p>
                    </div>
                    <div className="ih-job-actions">
                      <span className="ih-match">
                        {(rec.match_score * 100).toFixed(1)}% Match
                      </span>
                      {isAlreadyApplied ? (
                        <span className="ih-applied">✓ Applied</span>
                      ) : (
                        <button onClick={() => handleApply(rec)}>
                          Apply Now →
                        </button>
                      )}
                    </div>
                  </div>
                );
              })}
            </div>
          </section>
        )}

        {stipendStats.length > 0 && (
          <section className="ih-data-section">
            <h2>Role Stipend Analytics & Averages</h2>
            <div className="ih-stipend-grid">
              {stipendStats.map((stat, idx) => (
                <div className="ih-stipend-card" key={idx}>
                  <strong>{stat.role}</strong>
                  <p>Average: <span>₹{stat.average_stipend.toLocaleString()}</span></p>
                  <p>Median: ₹{stat.median_stipend.toLocaleString()}</p>
                  <small>Records analyzed: {stat.count}</small>
                </div>
              ))}
            </div>
          </section>
        )}

        {myApplications.length > 0 && (
          <section className="ih-data-section">
            <h2>My Applications ({myApplications.length})</h2>
            <p className="ih-muted">Application status and progress timeline</p>

            <div className="ih-application-list">
              {myApplications.map((app) => (
                <div className="ih-application-card" key={app.application_id}>
                  <div className="ih-application-heading">
                    <div>
                      <h3>{app.title}</h3>
                      <p>{app.company_name} · {app.opportunity_type}</p>
                    </div>
                    <span>{app.status}</span>
                  </div>

                  {app.history && app.history.length > 0 && (
                    <div className="ih-history">
                      <strong>Status Timeline</strong>
                      {app.history.map((h: any, idx: number) => (
                        <p key={idx}>
                          <span>●</span> {h.status} — {h.notes} (
                          {new Date(h.changed_at).toLocaleDateString()})
                        </p>
                      ))}
                    </div>
                  )}
                </div>
              ))}
            </div>
          </section>
        )}

        <footer className="ih-footer">
          <strong>✥ InternHub</strong>
          <span>Find opportunities that match your skills.</span>
        </footer>
      </div>

      <style jsx>{`
