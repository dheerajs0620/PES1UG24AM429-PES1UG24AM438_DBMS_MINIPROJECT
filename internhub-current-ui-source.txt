"use client";

import { useEffect, useRef, useState } from "react";
import { useRouter } from "next/navigation";

type User = {
  first_name?: string;
  last_name?: string;
  email?: string;
  role?: string;
};

type UploadStatus = "idle" | "extracting" | "saving" | "complete" | "error";

const categoryMap: Record<string, string[]> = {
  "Programming": [
    "Python", "C", "C++", "JavaScript", "TypeScript", "Java",
    "Rust", "React", "Next.js", "Node.js", "Express.js", "FastAPI"
  ],
  "Databases": [
    "MySQL", "PostgreSQL", "MongoDB", "Neo4j", "Redis", "ChromaDB",
    "SQL", "Database Management"
  ],
  "AI / ML": [
    "Machine Learning", "Deep Learning", "Computer Vision",
    "Natural Language Processing", "PyTorch", "TensorFlow", "OpenCV"
  ],
  "Other Skills": [
    "Git", "Docker", "Kubernetes", "Wazuh", "Linux",
    "AWS", "Communication", "Problem Solving"
  ]
};

export default function UploadPage() {
  const router = useRouter();
  const inputRef = useRef<HTMLInputElement>(null);

  const [user, setUser] = useState<User | null>(null);
  const [authChecked, setAuthChecked] = useState(false);
  const [file, setFile] = useState<File | null>(null);
  const [skills, setSkills] = useState<string[]>([]);
  const [status, setStatus] = useState<UploadStatus>("idle");
  const [message, setMessage] = useState("");
  const [dragging, setDragging] = useState(false);
  const [dropdownOpen, setDropdownOpen] = useState(false);

  useEffect(() => {
    let active = true;

    async function loadUser() {
      try {
        const response = await fetch("/api/auth/me", {
          credentials: "include",
          cache: "no-store"
        });

        const data = await response.json();

        if (active && data.success && data.user) {
          setUser(data.user);
        }
      } catch (error) {
        console.error("Unable to load user:", error);
      } finally {
        if (active) setAuthChecked(true);
      }
    }

    loadUser();

    return () => {
      active = false;
    };
  }, []);

  const userName = user
    ? `${user.first_name || ""} ${user.last_name || ""}`.trim()
    : "Student";

  const fileSize = (bytes: number) => {
    return bytes >= 1024 * 1024
      ? `${(bytes / (1024 * 1024)).toFixed(2)} MB`
      : `${(bytes / 1024).toFixed(0)} KB`;
  };

  const chooseFile = (selected: File | null) => {
    if (!selected) return;

    if (!selected.name.toLowerCase().endsWith(".pdf")) {
      setMessage("Please upload a PDF file.");
      setStatus("error");
      return;
    }

    if (selected.size > 5 * 1024 * 1024) {
      setMessage("File size must be 5 MB or less.");
      setStatus("error");
      return;
    }

    if (selected.size === 0) {
      setMessage("The selected file is empty.");
      setStatus("error");
      return;
    }

    setFile(selected);
    setSkills([]);
    setStatus("idle");
    setMessage("");
  };

  const extractPdfText = async (selectedFile: File): Promise<string> => {
    const pdfjs: any = await import("pdfjs-dist/build/pdf");

    pdfjs.GlobalWorkerOptions.workerSrc = "/pdf.worker.min.js";

    const buffer = await selectedFile.arrayBuffer();
    const pdf = await pdfjs.getDocument({ data: buffer }).promise;

    let fullText = "";

    for (let pageNumber = 1; pageNumber <= pdf.numPages; pageNumber++) {
      const page = await pdf.getPage(pageNumber);
      const content = await page.getTextContent();

      const pageText = content.items
        .map((item: any) => item.str || "")
        .join(" ");

      fullText += pageText + "\n";
    }

    if (!fullText.trim()) {
      throw new Error(
        "No readable text was found. This PDF may be scanned or image-based."
      );
    }

    return fullText;
  };

  const handleExtract = async () => {
    if (!file) {
      setMessage("Choose a resume before extracting skills.");
      setStatus("error");
      return;
    }

    if (!user) {
      setMessage("Please log in before uploading your resume.");
      setStatus("error");
      return;
    }

    try {
      setStatus("extracting");
      setMessage("Reading your resume and extracting skills...");

      const cvText = await extractPdfText(file);

      const extractResponse = await fetch("/api/extract-cv", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          cvText,
          fileName: file.name
        })
      });

      const extractData = await extractResponse.json();

      if (!extractResponse.ok || !extractData.success) {
        throw new Error(
          extractData.error || "Unable to extract skills from your resume."
        );
      }

      const extractedSkills: string[] =
        extractData.extracted_skills || [];

      if (extractedSkills.length === 0) {
        throw new Error(
          "No recognized skills were found. Please check your resume."
        );
      }

      setSkills(extractedSkills);
      setStatus("saving");
      setMessage("Saving your resume and extracted skills...");

      const saveResponse = await fetch("/api/resumes", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        credentials: "include",
        body: JSON.stringify({
          fileName: file.name,
          extractedText: cvText,
          extractedSkills
        })
      });

      const saveData = await saveResponse.json();

      if (!saveResponse.ok || !saveData.success) {
        throw new Error(
          saveData.error || "Skills were extracted but resume saving failed."
        );
      }

      setStatus("complete");
      setMessage("Resume analyzed and saved successfully!");
    } catch (error: any) {
      console.error("Resume processing error:", error);
      setStatus("error");
      setMessage(error.message || "Something went wrong. Please try again.");
    }
  };

  const removeFile = () => {
    setFile(null);
    setSkills([]);
    setStatus("idle");
    setMessage("");

    if (inputRef.current) {
      inputRef.current.value = "";
    }
  };

  const groupedSkills = Object.entries(categoryMap)
    .map(([category, categorySkills]) => ({
      category,
      skills: skills.filter((skill) =>
        categorySkills.some(
          (item) => item.toLowerCase() === skill.toLowerCase()
        )
      )
    }))
    .filter((group) => group.skills.length > 0);

  const ungroupedSkills = skills.filter(
    (skill) =>
      !Object.values(categoryMap).flat().some(
        (item) => item.toLowerCase() === skill.toLowerCase()
      )
  );

  const allGroups = [
    ...groupedSkills,
    ...(ungroupedSkills.length
      ? [{ category: "Additional Skills", skills: ungroupedSkills }]
      : [])
  ];

  return (
    <main className="upload-page">
      <header className="topbar">
        <a href="/" className="brand">
          <span className="brand-icon">✥</span>
          <span>InternHub</span>
        </a>

        <nav className="nav-links">
          <a href="/">Home</a>
          <a href="/recommendations">Jobs</a>
          <a href="/companies">Companies</a>
          <a href="/stipend-insights">Stipend Insights</a>
          <a href="/about">About</a>
        </nav>

        <div className="profile-area">
          {authChecked && user ? (
            <>
              <button
                className="profile-button"
                onClick={() => setDropdownOpen(!dropdownOpen)}
              >
                <span className="avatar">
                  {userName.charAt(0).toUpperCase()}
                </span>
                <span>Student</span>
                <span className="chevron">⌄</span>
              </button>

              {dropdownOpen && (
                <div className="profile-dropdown">
                  <strong>{userName}</strong>
                  <span>{user.email}</span>
                  <a href="/recommendations">My Jobs</a>
                  <a href="/profile">My Profile</a>
                </div>
              )}
            </>
          ) : (
            <a className="login-link" href="/login">
              Login
            </a>
          )}
        </div>
      </header>

      <div className="page-container">
        <section className="hero">
          <div className="hero-copy">
            <div className="eyebrow">
              <span className="eyebrow-dot" />
              YOUR CAREER STARTS HERE
            </div>

            <h1>Upload Your Resume</h1>

            <p>
              Get personalized internship recommendations based on your
              skills.
            </p>
          </div>

          <div className="hero-illustration" aria-hidden="true">
            <div className="glow-orb orb-one" />
            <div className="glow-orb orb-two" />

            <div className="resume-art">
              <div className="resume-art-title">RESUME</div>
              <div className="resume-art-avatar">♙</div>
              <div className="resume-art-line long" />
              <div className="resume-art-line medium" />
              <div className="resume-art-line short" />
              <div className="resume-art-line medium" />
            </div>

            <div className="cloud-art">↑</div>
          </div>
        </section>

        <section className="steps">
          <div className="step active">
            <div className="step-circle">1</div>
            <span>Upload Resume</span>
          </div>

          <div className="step-line">
            <span />
          </div>

          <div className={`step ${status === "complete" ? "active" : ""}`}>
            <div className="step-circle">2</div>
            <span>Extract Skills</span>
          </div>

          <div className="step-line">
            <span />
          </div>

          <div className="step">
            <div className="step-circle">3</div>
            <span>View Recommendations</span>
          </div>
        </section>

        <section className="main-grid">
          <div className="left-column">
            <div
              className={`upload-card ${dragging ? "dragging" : ""}`}
              onDragOver={(event) => {
                event.preventDefault();
                setDragging(true);
              }}
              onDragLeave={() => setDragging(false)}
              onDrop={(event) => {
                event.preventDefault();
                setDragging(false);
                chooseFile(event.dataTransfer.files[0] || null);
              }}
            >
              <input
                ref={inputRef}
                type="file"
                accept=".pdf,application/pdf"
                hidden
                onChange={(event) =>
                  chooseFile(event.target.files?.[0] || null)
                }
              />

              <div className="upload-icon">
                <svg viewBox="0 0 48 48" fill="none">
                  <path
                    d="M24 30V5m0 0L14 15M24 5l10 10M8 28v12h32V28"
                    stroke="currentColor"
                    strokeWidth="3"
                    strokeLinecap="round"
                    strokeLinejoin="round"
                  />
                </svg>
              </div>

              <h2>Drag and drop your resume here</h2>
              <p>or click to browse</p>
              <span className="upload-hint">
                Supports PDF files (Max 5MB)
              </span>

              <button
                className="primary-button choose-button"
                onClick={() => inputRef.current?.click()}
              >
                <span>▣</span> Choose File
              </button>
            </div>

            {file && (
              <div className="file-card">
                <div className="pdf-icon">
                  <span>PDF</span>
                </div>

                <div className="file-info">
                  <strong>{file.name}</strong>
                  <span>{fileSize(file.size)}</span>
                </div>

                <div className="file-status">
                  {status === "complete" ? (
                    <>
                      <span className="success-check">✓</span>
                      <span>Upload complete</span>
                    </>
                  ) : status === "extracting" || status === "saving" ? (
                    <>
                      <span className="spinner" />
                      <span>
                        {status === "extracting" ? "Analyzing..." : "Saving..."}
                      </span>
                    </>
                  ) : (
                    <span className="pending-label">Ready</span>
                  )}
                </div>

                <button
                  className="remove-button"
                  onClick={removeFile}
                  title="Remove file"
                  disabled={status === "extracting" || status === "saving"}
                >
                  ♧
                </button>
              </div>
            )}

            {message && (
              <div
                className={`message ${
                  status === "error" ? "error-message" :
                  status === "complete" ? "success-message" : ""
                }`}
              >
                {status === "complete" ? "✓" : status === "error" ? "!" : "◌"}
                <span>{message}</span>
              </div>
            )}

            <button
              className="primary-button extract-button"
              onClick={handleExtract}
              disabled={!file || status === "extracting" || status === "saving"}
            >
              {status === "extracting" || status === "saving" ? (
                <>
                  <span className="spinner" />
                  Processing Resume...
                </>
              ) : status === "complete" ? (
                <>✓ Resume Analyzed Successfully</>
              ) : (
                <>✦ Extract Skills from Resume <span>→</span></>
              )}
            </button>

            {status === "complete" && (
              <button
                className="continue-button"
                onClick={() => router.push("/recommendations")}
              >
                View Internship Recommendations →
              </button>
            )}
          </div>

          <aside className="right-column">
            <div className="info-card">
              <h3>How it Works?</h3>

              <div className="info-step">
                <div className="info-icon">▤</div>
                <div>
                  <strong>Upload Resume</strong>
                  <p>Add your latest resume (PDF).</p>
                </div>
              </div>

              <div className="info-connector">↓</div>

              <div className="info-step">
                <div className="info-icon">⚙</div>
                <div>
                  <strong>Extract Skills</strong>
                  <p>We analyze your resume and identify key skills, tools and domains.</p>
                </div>
              </div>

              <div className="info-connector">↓</div>

              <div className="info-step">
                <div className="info-icon">◎</div>
                <div>
                  <strong>Get Recommendations</strong>
                  <p>Receive internship matches tailored to your skills.</p>
                </div>
              </div>
            </div>

            <div className="info-card features-card">
              <h3>Supported Features</h3>

              <div className="feature-row">
                <span>✓</span> Automatic skill extraction
              </div>
              <div className="feature-row">
                <span>✓</span> Domain and role classification
              </div>
              <div className="feature-row">
                <span>✓</span> Personalized internship matching
              </div>
              <div className="feature-row">
                <span>✓</span> Match score with skill gap analysis
              </div>
            </div>
          </aside>
        </section>

        <section className="skills-card">
          <div className="skills-heading">
            <div>
              <h2>Extracted Skills Preview</h2>
              <p>
                {skills.length > 0
                  ? "Skills identified from your uploaded resume"
                  : "Your extracted skills will appear here after analysis"}
              </p>
            </div>

            <div className="skills-count">
              <strong>{skills.length}</strong>
              <span>Skills Found</span>
            </div>
          </div>

          {skills.length > 0 ? (
            <div className="skills-content">
              <div className="skills-ring">
                <div className="ring-inner">
                  <strong>{skills.length}</strong>
                  <span>Skills Found</span>
                </div>
              </div>

              <div className="skills-groups">
                {allGroups.map((group) => (
                  <div className="skill-group" key={group.category}>
                    <div className="group-title">
                      <span className="group-symbol">
                        {group.category === "Programming" ? "</>" :
                         group.category === "Databases" ? "▤" :
                         group.category === "AI / ML" ? "✺" : "★"}
                      </span>
                      <strong>{group.category}</strong>
                    </div>

                    <div className="skill-tags">
                      {group.skills.map((skill) => (
                        <span className="skill-tag" key={skill}>
                          {skill}
                        </span>
                      ))}
                    </div>
                  </div>
                ))}
              </div>
            </div>
          ) : (
            <div className="empty-skills">
              <div className="empty-icon">✧</div>
              <span>Upload and analyze your resume to see your skills here.</span>
            </div>
          )}
        </section>

        <footer className="page-footer">
          <span>✥ InternHub</span>
          <span>Find opportunities that match your skills.</span>
        </footer>
      </div>

      <style jsx>{`
        * {
          box-sizing: border-box;
        }

        .upload-page {
          min-height: 100vh;
          background:
            radial-gradient(ellipse at 80% 10%, rgba(91, 33, 182, .13), transparent 35%),
            radial-gradient(ellipse at 5% 60%, rgba(30, 64, 175, .10), transparent 35%),
            #050817;
          color: #f8fafc;
          font-family: Arial, Helvetica, sans-serif;
          overflow-x: hidden;
        }

        .topbar {
          height: 70px;
          padding: 0 5.5%;

              <option value="low-high">Match: Low to High</option>
            </select>

            <select
              className="ih-filter-select"
              value={typeFilter}
              onChange={(event) => {
                setTypeFilter(event.target.value);
                setCurrentPage(1);
              }}
            >
              <option value="all">Type: All</option>
              <option value="internship">Internship</option>
              <option value="full-time">Full-time</option>
              <option value="part-time">Part-time</option>
              <option value="project">Project</option>
            </select>

            <select
              className="ih-filter-select"
              value={modeFilter}
              onChange={(event) => {
                setModeFilter(event.target.value);
                setCurrentPage(1);
              }}
            >
              <option value="all">Mode: All</option>
              <option value="remote">Remote</option>
              <option value="hybrid">Hybrid</option>
              <option value="onsite">On-site</option>
            </select>

            <button
              type="button"
              className="ih-reset-button"
              onClick={() => {
                setSearchTerm('');
                setSortOrder('high-low');
                setTypeFilter('all');
                setModeFilter('all');
                setCurrentPage(1);
              }}
            >
              ↻ Reset
            </button>
          </div>

          <div className="ih-muted" style={{ marginBottom: 12 }}>
            Showing {visibleRecommendations.length} of {filteredRecommendations.length} opportunities
          </div>

          <div className="ih-opportunity-list">
            {visibleRecommendations.length === 0 ? (
              <div className="ih-empty-state">
                {recommendations.length === 0
                  ? 'Upload your resume and extract your skills to discover matching internships.'
                  : 'No opportunities match your search or filters.'}
              </div>
            ) : (
              visibleRecommendations.map((opportunity: any, index: number) => {
                const score = getMatchPercent(opportunity);
                const color = getMatchColor(score);
                const breakdown = getBreakdown(opportunity);
                const id = getOpportunityId(opportunity);
                const company = opportunity.company_name || opportunity.company || 'Company';
                const skills = getSkillList(opportunity);

                return (
                  <article className="ih-opportunity-card" key={id ?? index}>
                    <div className="ih-company-logo">
                      {opportunity.company_logo ? (
                        <img src={opportunity.company_logo} alt={`${company} logo`} />
                      ) : (
                        getCompanyInitial(company)
                      )}
                    </div>

                    <div className="ih-opportunity-info">
                      <h3>{opportunity.title || opportunity.role || 'Internship Opportunity'}</h3>
                      <div className="ih-company-name">{company}</div>

                      <div className="ih-opportunity-meta">
                        <span>⌖ {opportunity.location || 'Location not specified'}</span>
                        <span>▣ {getOpportunityDuration(opportunity)}</span>
                        <span>₹ {getOpportunityStipend(opportunity)}</span>
                      </div>

                      <div className="ih-opportunity-skills">
                        {skills.slice(0, 5).map((skill: string) => (
                          <span className="ih-opportunity-skill" key={skill}>
                            {skill}
                          </span>
                        ))}
                      </div>
                    </div>

                    <div className="ih-match-column">
                      <div
                        className="ih-match-ring"
                        style={{
                          background: `conic-gradient(${color} ${score * 3.6}deg, #293653 0deg)`,
                        }}
                      >
                        <div className="ih-match-ring-inner">
                          <strong>{Math.round(score)}%</strong>
                          <span>Match</span>
                        </div>
                      </div>
                    </div>

                    <div className="ih-breakdown">
                      <div className="ih-breakdown-title">Match Breakdown</div>

                      {[
                        ['Skill Match', breakdown.skill],
                        ['Role Fit', breakdown.role],
                        ['Domain Fit', breakdown.domain],
                      ].map(([label, value]) => (
                        <div className="ih-breakdown-row" key={String(label)}>
                          <span>{label}</span>
                          <div className="ih-progress-track">
                            <div
                              className="ih-progress-fill"
                              style={{
                                width: `${Math.max(0, Math.min(100, Number(value)))}%`,
                                background: color,
                              }}
                            />
                          </div>
                          <strong>{Math.round(Number(value))}%</strong>
                        </div>
                      ))}
                    </div>

                    <div className="ih-card-actions">
                      <button
                        type="button"
                        className={`ih-save-button ${savedOpportunityIds.includes(id) ? 'saved' : ''}`}
                        title={savedOpportunityIds.includes(id) ? 'Unsave opportunity' : 'Save opportunity'}
                        onClick={() => toggleSavedOpportunity(id)}
                      >
                        {savedOpportunityIds.includes(id) ? '♥' : '♧'}
                      </button>

                      <button
                        type="button"
                        className="ih-view-button"
                        onClick={() => setSelectedOpportunity(opportunity)}
                      >
                        View Details →
                      </button>
                    </div>
                  </article>
                );
              })
            )}
          </div>

          {totalPages > 1 && (
            <div className="ih-pagination">
              <button
                type="button"
                className="ih-page-button"
                disabled={currentPage === 1}
                onClick={() => setCurrentPage((page) => Math.max(1, page - 1))}
              >
                ‹
              </button>

              {Array.from({ length: totalPages }, (_, index) => index + 1).map((page) => (
                <button
                  type="button"
                  key={page}
                  className={`ih-page-button ${currentPage === page ? 'active' : ''}`}
                  onClick={() => setCurrentPage(page)}
                >
                  {page}
                </button>
              ))}

              <button
                type="button"
                className="ih-page-button"
                disabled={currentPage === totalPages}
                onClick={() => setCurrentPage((page) => Math.min(totalPages, page + 1))}
              >
                ›
              </button>
            </div>
          )}
        </section>
                {/* FULL INTERNSHIP DETAILS VIEW */}
        {selectedOpportunity && (
          <div className="ih-job-detail-overlay">
            <div className="ih-job-detail-page">

              <button
                type="button"
                className="ih-back-jobs"
                onClick={() => {
                  setSelectedOpportunity(null);
                  setDetailTab('overview');
                }}
              >
                ← Back to jobs
              </button>

              {/* JOB HEADER */}
              <div className="ih-job-header">

                <div className="ih-job-company-logo">
                  {selectedOpportunity.company_logo ? (
                    <img
                      src={selectedOpportunity.company_logo}
                      alt="Company logo"
                    />
                  ) : (
                    getCompanyInitial(
                      selectedOpportunity.company_name ||
                      selectedOpportunity.company ||
                      'Company'
                    )
                  )}
                </div>

                <div className="ih-job-heading">
                  <h1>
                    {selectedOpportunity.title ||
                      selectedOpportunity.role ||
                      'Internship Opportunity'}
                  </h1>

                  <div className="ih-job-badges">
                    <span className="ih-job-match-badge">
                      {Math.round(getMatchPercent(selectedOpportunity))}% Match
                    </span>

                    <span className="ih-job-featured-badge">
                      Featured
                    </span>
                  </div>

                  <div className="ih-job-meta">
                    <span>📍 {selectedOpportunity.location || 'Remote'}</span>
                    <span>▣ {getOpportunityDuration(selectedOpportunity)}</span>
                    <span>▣ ₹{getOpportunityStipend(selectedOpportunity)}</span>
                    <span>
                      ◷ Posted{' '}
                      {selectedOpportunity.posted_at
                        ? new Date(selectedOpportunity.posted_at).toLocaleDateString()
                        : 'Recently'}
                    </span>
                  </div>
                </div>

                <div className="ih-job-header-actions">
                  <button
                    type="button"
                    className="ih-job-apply-button"
                    onClick={() => handleApply(selectedOpportunity)}
                  >
                    Apply Now
                  </button>

                  <button
                    type="button"
                    className={`ih-job-save-button ${
                      savedOpportunityIds.includes(
                        getOpportunityId(selectedOpportunity)
                      ) ? 'saved' : ''
                    }`}
                    onClick={() =>
                      toggleSavedOpportunity(
                        getOpportunityId(selectedOpportunity)
                      )
                    }
                  >
                    ♧ &nbsp;
                    {savedOpportunityIds.includes(
                      getOpportunityId(selectedOpportunity)
                    ) ? 'Saved Job' : 'Save Job'}
                  </button>
                </div>

              </div>

              {/* DETAIL TABS */}
              <div className="ih-job-tabs">
                <button
                  type="button"
                  className={detailTab === 'overview' ? 'active' : ''}
                  onClick={() => setDetailTab('overview')}
                >
                  Overview
                </button>

                <button
                  type="button"
                  className={detailTab === 'requirements' ? 'active' : ''}
                  onClick={() => setDetailTab('requirements')}
                >
                  Requirements
                </button>

                <button
                  type="button"
                  className={detailTab === 'company' ? 'active' : ''}
                  onClick={() => setDetailTab('company')}
                >
                  About Company
                </button>
              </div>

              {/* OVERVIEW TAB */}
              {detailTab === 'overview' && (
                <div className="ih-job-detail-grid">

                  <div className="ih-job-detail-card">
                    <h2>Job Description</h2>

                    <p className="ih-job-description">
                      {selectedOpportunity.description ||
                        'Join the team and work on exciting projects while developing your technical and professional skills.'}
                    </p>

                    <h2 className="ih-responsibilities-heading">
                      Key Responsibilities
                    </h2>

                    <ul className="ih-responsibility-list">
                      {(Array.isArray(selectedOpportunity.responsibilities)
                        ? selectedOpportunity.responsibilities
                        : [
                            'Work on real-world projects and assigned tasks',
                            'Collaborate with team members and developers',
                            'Apply technical skills to solve practical problems',
                            'Participate in reviews and team discussions',
                          ]
                      ).map((item: string, index: number) => (
                        <li key={index}>
                          <span>●</span>
                          {item}
                        </li>
                      ))}
                    </ul>
                  </div>

                  <div className="ih-job-detail-right">

                    <div className="ih-job-detail-card">
                      <h2>Required Skills</h2>

                      <div className="ih-detail-skill-list">
                        {getSkillList(selectedOpportunity).length > 0 ? (
                          getSkillList(selectedOpportunity).map(
                            (skill: string) => (
                              <span key={skill}>{skill}</span>
                            )
                          )
                        ) : (
                          <span>Skills not specified</span>
                        )}
                      </div>

                      <h2 className="ih-preferred-heading">
                        Preferred Skills
                      </h2>

                      <div className="ih-detail-skill-list preferred">
                        {(Array.isArray(selectedOpportunity.preferred_skills)
                          ? selectedOpportunity.preferred_skills
                          : []
                        ).map((skill: string) => (
                          <span key={skill}>{skill}</span>
                        ))}

                        {!Array.isArray(selectedOpportunity.preferred_skills) && (
                          <span>Not specified</span>
                        )}
                      </div>
                    </div>

                    <div className="ih-job-detail-card ih-match-analysis">
                      <h2>Match Analysis</h2>

                      {(() => {
                        const required = getSkillList(selectedOpportunity);
                        const candidateSkills = extractedSkills.map(
                          (skill: any) =>
                            String(
                              typeof skill === 'string'
                                ? skill
                                : skill.name || skill.skill_name || ''
                            ).toLowerCase()
                        );

                        const matched = required.filter((skill: string) =>
                          candidateSkills.includes(skill.toLowerCase())
                        );

                        const missing = required.filter(
                          (skill: string) =>
                            !candidateSkills.includes(skill.toLowerCase())
                        );

                        const preferred = Array.isArray(
                          selectedOpportunity.preferred_skills
                        )
                          ? selectedOpportunity.preferred_skills
                          : [];

                        return (
                          <div className="ih-match-analysis-content">

                            <div
                              className="ih-large-match-ring"
                              style={{
                                background: `conic-gradient(#14e89b ${
                                  getMatchPercent(selectedOpportunity) * 3.6
                                }deg, #273752 0deg)`,
                              }}
                            >
                              <div>
                                <strong>
                                  {Math.round(
                                    getMatchPercent(selectedOpportunity)
                                  )}%
                                </strong>
                              </div>
                            </div>

                            <div className="ih-match-legend">
                              <div>
                                <span className="ih-legend-dot matched" />
                                <span>Matched Skills</span>
                                <strong>
                                  {matched.length}/{required.length}
                                </strong>
                              </div>

                              <div>
                                <span className="ih-legend-dot missing" />
                                <span>Missing Skills</span>
                                <strong>
                                  {missing.length}/{required.length}
                                </strong>
                              </div>

                              <div>
                                <span className="ih-legend-dot optional" />
                                <span>Optional Skills</span>
                                <strong>
                                  {preferred.length}
                                </strong>
                              </div>
                            </div>

                          </div>
                        );
                      })()}
                    </div>

                  </div>
                </div>
              )}

              {/* REQUIREMENTS TAB */}
              {detailTab === 'requirements' && (
                <div className="ih-job-detail-grid">

                  <div className="ih-job-detail-card">
                    <h2>Required Skills</h2>

                    <p className="ih-job-description">
                      These are the skills associated with this opportunity.
                    </p>

                    <div className="ih-detail-skill-list">
                      {getSkillList(selectedOpportunity).map(
                        (skill: string) => (
                          <span key={skill}>{skill}</span>
                        )
                      )}
                    </div>

                    <h2 className="ih-preferred-heading">
                      Preferred Skills
                    </h2>

                    <div className="ih-detail-skill-list preferred">
                      {(Array.isArray(selectedOpportunity.preferred_skills)
                        ? selectedOpportunity.preferred_skills
                        : []
                      ).map((skill: string) => (
                        <span key={skill}>{skill}</span>
                      ))}
                    </div>
                  </div>

                  <div className="ih-job-detail-card">
                    <h2>Your Skill Compatibility</h2>
                    <p className="ih-job-description">
                      Your resume has been compared with the skills required
                      for this opportunity.
                    </p>

                    <div className="ih-requirement-score">
                      <strong>
                        {Math.round(getMatchPercent(selectedOpportunity))}%
                      </strong>
                      <span>Overall Match</span>
                    </div>
                  </div>

                </div>
              )}

              {/* ABOUT COMPANY TAB */}
              {detailTab === 'company' && (
                <div className="ih-job-detail-card ih-company-about">
                  <h2>
                    About{' '}
                    {selectedOpportunity.company_name ||
                      selectedOpportunity.company ||
                      'Company'}
                  </h2>

                  <p className="ih-job-description">
                    {selectedOpportunity.company_description ||
                      'Company information has not been provided for this opportunity yet.'}
                  </p>

                  <div className="ih-company-detail-grid">
                    <div>
                      <span>Company</span>
                      <strong>
                        {selectedOpportunity.company_name ||
                          selectedOpportunity.company ||
                          'Not specified'}
                      </strong>
                    </div>

                    <div>
                      <span>Location</span>
                      <strong>
                        {selectedOpportunity.location || 'Not specified'}
                      </strong>
                    </div>

                    <div>
                      <span>Website</span>
                      <strong>
                        {selectedOpportunity.company_website || 'Not provided'}
                      </strong>
                    </div>
                  </div>
                </div>
              )}

            </div>
          </div>
        )}

        {/* APPLICATION FEEDBACK */}
        {applyMsg && (
          <div className="ih-application-feedback">
            {applyMsg}
            <button
              type="button"
              onClick={() => setApplyMsg('')}
            >
              ×
            </button>
          </div>
        )}
                

      </div>
    </main>
  );
}
